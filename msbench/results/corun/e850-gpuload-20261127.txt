# gpuload.sh (11.118): CPU worker ncnn fp16 4 threads CPU4-7 for 30 s, alone / with gltest load fma|copy on CPU0
cd ~/msbench && export PYTHONPATH=py/3.11
cpuw() { OMP_PROC_BIND=close OMP_PLACES=cores taskset -c 4-7 python3 e850gpu/corun.py worker ncnn fp16 $1 4 1 30 $(date +%s) 2>/dev/null | tail -1; }
load() { ( . e850gpu/gpu-env.sh; taskset -c 0 e850gpu/gltest load $1 40 2>&1 | grep "^load" ) & }
for m in resnet18 yolo11n-320 mobilenetv2; do
  echo "== $m alone";      cpuw $m
  echo "== $m + gpu fma";  load fma;  sleep 5; cpuw $m; wait
  echo "== $m + gpu copy"; load copy; sleep 5; cpuw $m; wait
  echo "== $m alone";      cpuw $m
done
echo LOADDONE

# gpuload-clk.sh: resnet18 CPU worker 20 s, samples every 0.5 s: cpu4 kHz, cpu0 kHz, LITTLE1 mC, G3D mC, gpu kHz
cd ~/msbench && export PYTHONPATH=py/3.11
samp() { i=0; while [ $i -lt 36 ]; do echo "$(cat /sys/devices/system/cpu/cpu4/cpufreq/scaling_cur_freq) $(cat /sys/devices/system/cpu/cpu0/cpufreq/scaling_cur_freq) $(cat /sys/class/thermal/thermal_zone1/temp) $(cat /sys/class/thermal/thermal_zone2/temp) $(cat /sys/devices/platform/11500000.mali/clock 2>/dev/null || cat /sys/class/misc/mali0/device/clock)"; sleep 0.5; i=$((i+1)); done > /var/tmp/clk-$1.txt; }
cpuw() { OMP_PROC_BIND=close OMP_PLACES=cores taskset -c 4-7 python3 e850gpu/corun.py worker ncnn fp16 resnet18 4 1 20 $(date +%s) 2>/dev/null | tail -1; }
echo "== alone"; ( sleep 1; samp alone ) & cpuw; wait
( . e850gpu/gpu-env.sh; taskset -c 0 e850gpu/gltest load fma 30 2>&1 | grep "^load" ) & sleep 5
echo "== fma"; ( sleep 1; samp fma ) & cpuw; wait
for f in alone fma; do echo "-- $f: cpu4 MHz, cpu0 MHz, LITTLE1, G3D, gpu"; sort /var/tmp/clk-$f.txt | uniq -c | sort -rn | head -6; done
